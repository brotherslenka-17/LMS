cmake_minimum_required(VERSION 3.16)
project(ModeratedLMS VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(LMS_COVERAGE "Instrument first-party business code for gcov" OFF)
option(LMS_SANITIZERS "Enable address and undefined behaviour sanitizers" OFF)
find_path(SODIUM_INCLUDE_DIR sodium.h HINTS ${SODIUM_ROOT}/include)
find_library(SODIUM_LIBRARY NAMES sodium libsodium HINTS ${SODIUM_ROOT}/lib)
if(NOT SODIUM_INCLUDE_DIR OR NOT SODIUM_LIBRARY)
  message(FATAL_ERROR "Install libsodium development files or set -DSODIUM_ROOT=<prefix>. See README.md")
endif()
add_library(sqlite3 STATIC vendor/sqlite3.c)
target_include_directories(sqlite3 PUBLIC vendor)
if(UNIX)
 target_link_libraries(sqlite3 PUBLIC ${CMAKE_DL_LIBS} pthread m)
endif()
file(READ migrations/001_initial.sql INITIAL_SCHEMA_TEXT)
file(READ migrations/002_class_extensions.sql CLASS_SCHEMA_TEXT)
configure_file(include/lms/Schema.hpp.in generated/Schema.hpp @ONLY)
add_library(lms_core src/domain/Domain.cpp src/application/Services.cpp src/infrastructure/SqliteRepository.cpp)
target_include_directories(lms_core PUBLIC include PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_include_directories(lms_core SYSTEM PUBLIC ${SODIUM_INCLUDE_DIR})
target_link_libraries(lms_core PUBLIC sqlite3 ${SODIUM_LIBRARY})
if(MSVC)
 target_compile_options(lms_core PRIVATE /W4)
else()
 target_compile_options(lms_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(LMS_COVERAGE AND NOT MSVC)
 target_compile_options(lms_core PRIVATE --coverage -O0 -g)
 target_link_options(lms_core PUBLIC --coverage)
endif()
if(LMS_SANITIZERS AND NOT MSVC)
 target_compile_options(lms_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(lms_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(lms src/ui/main.cpp)
target_link_libraries(lms PRIVATE lms_core)
target_compile_definitions(lms PRIVATE LMS_VERSION="${PROJECT_VERSION}")
include(CTest)
if(BUILD_TESTING)
 add_executable(lms_tests tests/test_lms.cpp)
 target_include_directories(lms_tests SYSTEM PRIVATE vendor)
 target_link_libraries(lms_tests PRIVATE lms_core)
 add_test(NAME lms_tests COMMAND lms_tests --reporter junit --out test-results.xml)
endif()
install(TARGETS lms RUNTIME DESTINATION bin)
option(LMS_BUILD_BENCHMARK "Build the synthetic 800-student benchmark" OFF)
if(LMS_BUILD_BENCHMARK)
 add_executable(lms_benchmark tests/benchmark.cpp)
 target_link_libraries(lms_benchmark PRIVATE lms_core)
endif()
